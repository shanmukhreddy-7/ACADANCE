cmake_minimum_required(VERSION 3.16)
project(AcadenceCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(acadence_core src/attendance.cpp src/assessments.cpp src/gpa.cpp src/teams.cpp src/planning.cpp src/photo_attendance.cpp)
target_include_directories(acadence_core PUBLIC include)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE acadence_core)
enable_testing()
add_test(NAME academic_core COMMAND core_tests)
